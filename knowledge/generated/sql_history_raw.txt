#  New  Categories   Specific  SE  WISE  Report    SELECT      sk.status_retailer_id  AS  retailer_id,      c.id  AS  distributor_id,      m.name  AS  category,      count(sk.lpn_number)  as  scan_count,      SUM(          CASE              WHEN  sqpm.box_calculation_uom  IS  NOT  NULL                   AND  sqpm.box_calculation_uom  >  0              THEN  sqpm.box_calculation_uom              WHEN  sk.uom  =  'B5'  THEN  0.5              ELSE  1          END      )  AS  current_category_box_count,      sqpm.uom,  
    u.name,      u.mobile_number,      md.shop_name  FROM  sku_inventories  sk  JOIN  users  u      ON  u.id  =  sk.status_retailer_id  JOIN  sku_products  sp      ON  sp.id  =  sk.product_id  JOIN  master_categories  m      ON  m.id  =  sp.categories  JOIN  executive_distributor_mappings  edm      ON  edm.distributor_id  =  u.primary_distributor_id  JOIN  sku_qr_points_maps  sqpm      ON  sqpm.sku_code  =  sk.sku_code     AND  sqpm.uom  =  sk.uom  JOIN  mechanic_details  md      ON  md.mechanic_id  =  u.id  join  companies  c  on   c.id  =  u.primary_distributor_id  WHERE  u.user_role  =  2    AND  edm.executive_id  =  53478    AND  sk.retailer_scanned_at  >=  '2026-04-01'    AND  sk.retailer_scanned_at  <  '2026-05-01'    AND  NOT  EXISTS  (          SELECT  1          FROM  sku_inventories  sk2          JOIN  sku_products  sp2              ON  sp2.id  =  sk2.product_id          WHERE  sk2.status_retailer_id  =  sk.status_retailer_id            AND  sp2.categories  =  sp.categories            AND  sk2.retailer_scanned_at  <  '2026-03-01'    )  GROUP  BY      sk.status_retailer_id,      sk.distributer_id,      m.name,      sk.uom,      u.name,      u.mobile_number,      md.shop_name  ORDER  BY      sk.status_retailer_id,  
    m.name;
